## 1. `templates/_shell.html`（全文）
导出宏：`shell_sidebar(active)`、`shell_topbar(title, subtitle, active)`、`shell_head()`、`shell_scripts()`。
⚠️ 踩坑记录：Jinja 的 `{% import %}` **只导出宏**，import 之外的文件级内容（`<style>`/`<script>`）
不会出现在导入方的输出里（实测：直接渲染 22045 字节，import 后只剩 6168 字节）。
所以骨架样式与行为也包成了宏，由页面在 `<head>` 与 `</body>` 前显式调用。

```jinja

   _shell.html —— 全站共享骨架（Jinja 宏 + 内联 CSS + 内联原生 JS）
   ----------------------------------------------------------------------------
   为什么单独一个文件：侧边菜单只有这一处定义。五个页面各写一遍 = 改一条菜单
   要改五处，必然腐化。页面侧只需要两行：

       {% import "_shell.html" as shell %}
       {{ shell.shell_sidebar('trend') }}
       {{ shell.shell_topbar('实时趋势', '桥梁受载趋势 · 阈值仪表盘', 'trend') }}

     shell_sidebar(active)            → 桌面固定侧栏 + 移动端抽屉（含遮罩）
     shell_topbar(title, subtitle, active) → 顶栏（页标题 + 副标题 + 右侧状态区）

   离线红线：不引入任何第三方 UI 库 / 图标库 / 字体 CDN；图标一律内联 SVG
   （单色、stroke=currentColor、18~20px）；JS 是原生 ES5 语法（现场老旧浏览器也能跑）。

   无障碍：
     <nav aria-label="主导航">、当前项 aria-current="page"、汉堡按钮 aria-expanded，

     · 桌面侧栏 240px，收起 64px；右缘 1px 浅灰分隔线

  };
```

```jinja
      setText($('statLast'), d.last_sample_time || '--');
      setText($('statSamples'), String(d.sample_count || 0));
      setText($('gaugeThreshold'), d.threshold);
      setText($('trendFoot'), d.last_sample_time || '--');
      setText($('trendSource'), srcText);

      /* ---- 仪表盘：整体重设（量程/刻度带/指针色按当前阈值与风险等级生成） ---- */
      if (HAS_ECHARTS) {
        var gaugeSig = [hasValue ? d.strain : 'na', d.threshold, risk.sev].join('/');
        if (gaugeSig !== chartSig.gauge) {
          chartSig.gauge = gaugeSig;
          gaugeChart.setOption(gaugeOption(d), { notMerge: true });
        }
      }
      setText($('gaugeValue'), hasValue ? d.strain.toFixed(2) : '--');

      /* ---- 趋势折线图（含 markLine 阈值同步） ---- */
      var hist = d.history || [];
      var times = hist.map(function (i) { return i.time; });
      var vals = hist.map(function (i) { return i.strain; });
      if (HAS_ECHARTS) {
        var lineSig = d.threshold + '|' + times.length + '|' + times[times.length - 1] + '|' + (vals[vals.length - 1] || 0);
        if (lineSig !== chartSig.line) {
          chartSig.line = lineSig;
          lineChart.setOption({
            xAxis: { data: times },
            series: [{ data: vals, markLine: {
              data: [{ yAxis: d.threshold, name: '阈值' }],
              label: { formatter: '阈值 ' + d.threshold } } }]
          });
        }

      setText($('trendSpan'), vals.length ? ('共 ' + vals.length + ' 个采样点') : '等待采样');
      var cc = $('chartConn');
      if (cc) {
        cc.className = 'conn idle';
        cc.textContent = vals.length === 0 ? '等待采样数据…' : '';
      }

      /* ---- 近段采样迷你面积图（仅取 history 截尾，不额外造数） ---- */
      var tail = hist.slice(-30);
      var mTimes = tail.map(function (i) { return i.time; });
      var mVals = tail.map(function (i) { return i.strain; });
      if (HAS_ECHARTS) {

        : '最近历史点');
      setNum($('miniCount'), d.sample_count || 0, { dec: 0, suffix: ' 次' });

      /* ---- 顶栏状态区（全站一致：连接状态 / 模拟徽章 / 待处置铃铛） ---- */
      var pending = alarms.filter(function (a) { return a.status === '未处理'; }).length;

      bg().setShellConn(isSim ? 'sim' : 'ok', isSim ? '模拟数据源 · 每秒刷新' : '数据连接正常');

  resizeChartsSoon();
  window.addEventListener('resize', resizeCharts);
  window.addEventListener('load', resizeCharts);          // 字体/图片加载完会再触发一次重排
```

```jinja
/* ====================================================================
   设备与数据源页脚本 —— 从 templates/dashboard.html 内联脚本搬家
   搬家内容：设备健康度更新分支（h.device_id / h.text / h.state / h.timeout_count /
             h.last_heartbeat / h.last_error 全部原样口径）、sourceBadge 的 sim/serial 文案、
             fmtUptime、setNum、每秒 refresh()（401 → /login）
   本页无图表：不加载 echarts，也不引用 #donut / #slValue / #slDot。
   红线：数值只来自 /api/data；设备离线一律显示 -- 并给出原因。
   ==================================================================== */
(function () {
  'use strict';

  /* ---------------- 0. 基础工具（与 dashboard 同口径） ---------------- */
  function $(id) { return document.getElementById(id); }
  function pad2(n) { return String(n).padStart(2, '0'); }
  function setText(el, txt) { if (el) el.textContent = txt; }
  function reduceMotion() {
    return !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  }
  /* 顶栏状态区由 _shell.html 的 shell_scripts() 提供，它在页面脚本【之后】执行，
     所以必须调用时再取（不能在脚本开头缓存，否则永远拿到空实现）。 */
  var BG_NOOP = { setShellSim: function () {}, setShellConn: function () {}, setShellBell: function () {} };
  function bg() { return window.BGShell || BG_NOOP; }

  function fmtUptime(sec) {
    sec = Math.max(0, Math.floor(sec || 0));
    return pad2(Math.floor(sec / 3600)) + ':' + pad2(Math.floor(sec % 3600 / 60)) + ':' + pad2(sec % 60);
  }

  /* 离线原因：只陈述后端给的事实（last_error / timeout_count），不臆测、不美化。
     timeout_count 仅在 > 0 时附加，避免出现「连续超时 0 次」这种反直觉的描述。 */
  function offlineReason(h) {
    var why = (h && h.last_error) ? String(h.last_error) : '未收到心跳';
    var tc = h ? Number(h.timeout_count) : NaN;
    if (isFinite(tc) && tc > 0) why += '（连续超时 ' + tc + ' 次）';
    return why;
  }

  /* ---------------- 数字 count-up（reduced-motion 下直接赋值） ---------------- */
  var numPrev = Object.create(null);
  var numAnims = Object.create(null);
  function setNum(el, value, opt) {
    if (!el) return;
    opt = opt || {};
    if (value === null || value === undefined || !isFinite(value)) { el.textContent = '--'; numPrev[el.id] = null; return; }
    var dec = opt.dec || 0;
    var fmt = opt.fmt || function (v) { return v.toFixed(dec) + (opt.suffix || ''); };
    var key = el.id || 'anon';
    var prev = numPrev[key];
    numPrev[key] = value;
    var target = fmt(value);
    var raf = window.requestAnimationFrame;
    if (reduceMotion() || !raf || prev === null || prev === undefined) { el.textContent = target; return; }
    if (Math.abs(prev - value) < Math.pow(10, -dec) / 2) { el.textContent = target; return; }
    if (numAnims[key]) { raf.call(window, numAnims[key]); numAnims[key] = null; }
    var from = prev, span = value - prev, t0 = null, DUR = opt.dur || 620;
    el.classList.add('flash');
    window.setTimeout(function () { el.classList.remove('flash'); }, 780);
    numAnims[key] = function step(ts) {
      if (t0 === null) t0 = ts;
      var p = Math.min(1, (ts - t0) / DUR);
      var e = 1 - Math.pow(1 - p, 3);               // easeOutCubic
      el.textContent = fmt(from + span * e);
      if (p < 1) { numAnims[key] = step; raf.call(window, step); } else { numAnims[key] = null; }
    };
    raf.call(window, numAnims[key]);
  }

  /* ---------------- 横幅：离线 / 接口异常（如实说明，不静默） ---------------- */
  function showNotice(msg) {
    var box = $('devNotice');
    if (!box) return;
    box.hidden = false;
    setText($('devNoticeText'), msg);
  }
  function hideNotice() {
    var box = $('devNotice');
    if (box) box.hidden = true;
  }

  /* ====================================================================
     主刷新循环：每 1 秒轮询 /api/data
     ==================================================================== */
  async function refresh() {
    try {
      var r = await fetch('/api/data');
      if (r.status === 401) { window.location.href = '/login'; return; }
      if (!r.ok) throw new Error('请求失败：' + r.status);
      var d = await r.json();
      var alarms = d.alarms || [];
      var h = d.device_health || {};
      var isOnline = h.state !== 'offline';
      var isSim = d.source === 'sim';
      var srcText = isSim ? '模拟器' : '串口（真实设备）';

      /* ---- 采集参数 ---- */
      setText($('devThreshold'), d.threshold);
      setNum($('intervalShow'), d.interval, { fmt: function (v) { return v + ' 秒'; } });
      if (isOnline) {
        setText($('lastSampleTime'), d.last_sample_time || '--');
        setText($('lastSampleNote'), '数据来自 /api/data');
      } else {
        /* 离线：当前读数列显示 --，离线前的最后一条记录如实附注，不冒充当前读数 */
        setText($('lastSampleTime'), '--');
        setText($('lastSampleNote'), d.last_sample_time
          ? ('离线前最后一次：' + d.last_sample_time)
          : '离线期间无新增采样');
      }
      setNum($('heroSamples'), d.sample_count || 0, { dec: 0, suffix: ' 次' });

      /* ---- 连接状态（#healthState 原样口径：h.text || d.sensor_status || --） ---- */
      var st = $('healthState');
      if (st) {
        st.textContent = h.text || d.sensor_status || '--';
        st.className = 'health-state dv-badge ' +
          (h.state === 'online' ? 'state-online' : (h.state === 'offline' ? 'state-offline' : 'state-unknown'));
      }
      setText($('healthError'), h.last_error || '无');
      setText($('sourceBadge'), '数据源：' + (isSim ? '模拟' : '串口（真实）'));
      var sb = $('sourceBadge');
      if (sb) sb.className = 'source-badge' + (isSim ? '' : ' serial');
      var cl = $('connLight');
      if (cl) cl.className = isOnline ? 'conn' : 'conn error';
      setText($('uptimeShow'), fmtUptime(d.uptime || 0));

      /* ---- 设备健康度（离线如实显示，不伪装在线） ---- */
      setText($('healthDevice'), h.device_id || d.device_id || '--');
      setText($('healthSource'), srcText);
      setText($('healthTimeout'), (h.timeout_count === null || h.timeout_count === undefined)
        ? '--' : h.timeout_count + ' 次');
      setText($('healthHeartbeat'), isOnline ? (h.last_heartbeat || '--') : '--');

      /* ---- 系统运行 ---- */
      setNum($('uptimeNum'), d.uptime || 0, { fmt: fmtUptime });
      setNum($('heroUptime'), d.uptime || 0, { fmt: fmtUptime });
      setText($('sampleCount'), (d.sample_count || 0) + ' 次');
      setText($('sensorStatus'), d.sensor_status || '--');

      /* ---- 页脚 ---- */
      setText($('footSource'), srcText);
      setText($('footClock'), d.last_sample_time || '--');
      setText($('footLast'), d.last_sample_time || '--');

      /* ---- 离线横幅：原因全部来自后端字段 ---- */
      if (isOnline) {
        hideNotice();
      } else {
        showNotice('设备离线（' + (h.device_id || d.device_id || '设备标识未知') + '）：' +
          '最近心跳与最后采样显示 --，不伪造在线状态。原因：' + offlineReason(h));
      }

      /* ---- 顶栏状态区（全站一致） ---- */
      var pending = alarms.filter(function (a) { return a.status === '未处理'; }).length;
      bg().setShellSim(isSim);
      bg().setShellConn(isSim ? 'sim' : 'ok', isSim ? '模拟数据源 · 每秒刷新' : '数据连接正常');
      bg().setShellBell(pending);
    } catch (e) {
      console.error(e);
      var clErr = $('connLight');
      if (clErr) clErr.className = 'conn error';
      showNotice('接口断开：无法获取 /api/data，页面显示的是上一次成功刷新的结果。');
      bg().setShellConn('error', '接口断开');
    }
  }

  /* ---------------- 启动 ---------------- */
  refresh();
  window.setInterval(refresh, 1000);
})();
</script>
```

## 9. 已知风险点 / 留给 P2、P3 的接入点

1. **3D 画布仍是深色**（`scene.js` 的 `renderer.setClearColor(0x070708, 1)`）：这是 P2「3D 浅色渲染」的范围，本轮刻意不动。
   因此 `/dashboard` 整页亮度均值会被画布拉低（实测侧栏 250.9 / 顶栏+指标带 191.6 / 画布区 136.4），
   验收口径按「面板与侧栏白底可读、画布区豁免」。P2 改 `clearColor` + 材质/光照/地形配色后此问题自然消失。
2. **3D 帧率仍写死 30fps、抗锯齿开、像素比 1.4**：P2 范围（`scene.js` 未改）。接入点已留干净：
   渲染循环、`resize()`、`setVisible()` 都不在模板侧，模板只负责给画布容器正确尺寸（实测 buffer = client 尺寸）。
3. **HUD 工具条无法移出画布**：`hud.js` 把视角预设/缩放/开关/复位挂在 `.twin-hud`（`inset:0`）上，
   本轮不改 `hud.js`/`mount.js`，所以桌面端用绝对定位把它「停靠」在画布底边居中（视觉上就是一条工具条），
   手机端改为贴画布底边的单行横向滚动（44px 触摸目标），并用 `.hero` 的 `padding-bottom` 预留空间。
   若 P2 想把它做成真正的独立工具条行，需要在 `hud.js` 里把 `bar` 挂到 `.twin-stage` 之外。
4. **旧 app.css 第 1–19 节仍是深色时代源码**（含 `#bigscreen` 作用域规则）：本轮按要求只追加不重写，
   因此文件里同时存在新旧两套（新的一套在末尾、特异性更高，实际生效）。
   真正的清理（删旧节）建议放在 P2 之后单独一轮，且必须先有全量视觉回归。
5. **`.twin-hud-info`（桥型信息卡）与 `.twin-hud-hint`（鼠标提示）在首页默认隐藏**，
   全屏观看时才显示 —— 与改造前 `#bigscreen` 作用域里的行为一致。
   若之后想让信息卡常驻，注意 `.twin-hud` 是 `inset:0`，必须同时给 `right/bottom:auto`，
   否则 `bottom` 会被解析成 0，整块被拉成画布同高的大白板（本轮实测踩过：1200×790 的白色面板盖住左半边画布）。
6. **手机端页面总高约 1167px（390×844）**：单列纵向滚动是需求本身（恢复自然滚动），
   画布 46vh + 指标带 + 工具条 + 模拟条 + 页脚依次排列，触摸目标全部 ≥44px。
7. **`app.py` 的 2 处模板上下文补齐**（`settings_page` / `history_page` 加 `user=`）是本轮唯一超出
   「只加页面路由」的改动，用于让共享骨架显示登录用户与退出入口。如果你要求严格零改动，
   这两处可以回退（代价：这两页侧栏用户名显示为空、但退出入口仍可用）。
